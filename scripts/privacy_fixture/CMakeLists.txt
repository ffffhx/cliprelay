cmake_minimum_required(VERSION 3.20)
project(cliprelay_privacy_decode C)
set(CMAKE_C_STANDARD 11)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static fixture dependencies" FORCE)
if(NOT APPLE OR NOT EXISTS "${MOONLIGHT_COMMON}/src/Limelight.h")
  message(FATAL_ERROR "Disposable Mac fixture requires the pinned Moonlight common source")
endif()
add_subdirectory("${MOONLIGHT_COMMON}" moonlight)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavutil)
add_executable(cliprelay-privacy-decode decode.c)
target_link_libraries(cliprelay-privacy-decode PRIVATE moonlight-common-c PkgConfig::FFMPEG)
target_compile_options(cliprelay-privacy-decode PRIVATE -Wall -Wextra -Werror)
