cmake_minimum_required(VERSION 3.20)
project(cliprelay_audio_decode C)
set(CMAKE_C_STANDARD 11)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static fixture dependencies" FORCE)
if(NOT APPLE OR NOT EXISTS "${MOONLIGHT_COMMON}/src/Limelight.h")
  message(FATAL_ERROR "Disposable Mac fixture requires the pinned Moonlight common source")
endif()
add_subdirectory("${MOONLIGHT_COMMON}" moonlight)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavutil)
pkg_check_modules(OPUS REQUIRED IMPORTED_TARGET opus)
add_executable(cliprelay-audio-decode ../privacy_fixture/decode.c)
target_compile_definitions(cliprelay-audio-decode PRIVATE CLIPRELAY_AUDIO_FIXTURE=1)
target_link_libraries(cliprelay-audio-decode PRIVATE moonlight-common-c PkgConfig::FFMPEG PkgConfig::OPUS)
target_compile_options(cliprelay-audio-decode PRIVATE -Wall -Wextra -Werror)
